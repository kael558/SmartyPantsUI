	}

		chatWindowOpen: false,
		inputMessage: "",
		messages: [],

		issueFormOpen: false,

		openSourcesIdx: null,
		sessionId:
			Math.random().toString(36).substring(2, 15) +
			Math.random().toString(36).substring(2, 15),

}

function saveState(state) {
	localStorage.setItem("chatState", JSON.stringify(state));
}

function loadScript(url) {
	return new Promise((resolve, reject) => {
		const script = document.createElement("script");
		script.src = url;
		script.onload = resolve;
		script.onerror = reject;
		document.head.appendChild(script);
	});
}

function openAppointmentForm(calendarUrl) {
	const appointmentForm = shadow.getElementById("appointment-form");
	appointmentForm.style.display = "flex";

	// Check if it's a valid calendly URL with the pattern calendly.com/{name}
	if (
		calendarUrl &&
		/calendly\.com\/[a-zA-Z0-9_-]+(?:\/[a-zA-Z0-9_-]+)?$/.test(calendarUrl)
	) {
		const calendarIframe = shadow.getElementById("calendar-iframe");
		calendarIframe.src = calendarUrl;
	} else {
		console.warn("Invalid Calendly URL format:", calendarUrl);
		// Optionally set a default Calendly URL or show an error message
	}
}

async function loadDependencies() {
	try {
		// Load Fredoka font

		/*const fontLink = document.createElement('link');
		fontLink.href = 'https://fonts.googleapis.com/css2?family=Fredoka:wght@300..700&display=swap';
		fontLink.rel = 'stylesheet';
		document.head.appendChild(fontLink);

		await document.fonts.ready;*/

		// Load marked.js
		//await loadScript('https://cdn.jsdelivr.net/npm/marked/marked.min.js');
		const { marked } = await import(
			"https://cdn.jsdelivr.net/npm/marked/lib/marked.esm.js"
		);
		markedFn = marked;

		// Create a custom renderer
		const renderer = new marked.Renderer();

		// Override the link renderer
		renderer.link = function ({ href, title, text }) {
			if (href && href.includes("calendly.com")) {
				// Pass the href to openAppointmentForm
				return `<a href="javascript:void(0)" onclick="openAppointmentForm('${href}')">${text}</a>`;
			}

			// Return normal link for non-calendly links
			return `<a href="${href}" target="_blank" rel="noopener noreferrer">${text}</a>`;
		};

		marked.setOptions({
			renderer: renderer,
			highlight: function (code, lang) {
				return code;
			},
			gfm: true,
			breaks: true,
			headerIds: true,
			mangle: false,
		});
		// Optionally load DOMPurify for sanitization
		await loadScript(
			"https://cdnjs.cloudflare.com/ajax/libs/dompurify/3.0.1/purify.min.js"
		);
	} catch (error) {
		console.error("Failed to load dependencies:", error);
	}
}

			setTimeout(() => {
				welcomeMessageElement.classList.add("animate");

				setTimeout(() => {
					let i = 0;

							typedMessage.textContent = message.substring(

								i + 1
							);
							i++;
						} else {
							clearInterval(typeInterval);

					"circleExit 0.3s ease-out forwards";

					welcomeMessageElement.style.display = "none";

				}, 300);
			}, 6000);
		}, 300);

	async function handleSendMessage() {
		const input = shadow.getElementById("message-input");
		const message = input.value.trim();
		const loadingSpinner = shadow.getElementById("loading-spinner");
		const sendButton = shadow.getElementById("send-button");

		if (message) {
			// Disable input and show loading state
			input.disabled = true;
			sendButton.disabled = true;
			loadingSpinner.style.display = "flex";

			// Add user message to state
			state.messages.push({
				role: "user",
				content: message,
				timestamp: new Date().toISOString(),
			});

			saveState(state);

			// Create and append user message
			const messageDiv = createMessageElement(
				{
					role: "user",
					content: message,
					timestamp: new Date().toISOString(),
				},
				state.messages.length - 1
			);

			const messagesContainer = shadow.getElementById("chat-messages");
			messagesContainer.appendChild(messageDiv);
			messagesContainer.scrollTop = messagesContainer.scrollHeight;

			input.value = "";
			state.inputMessage = "";
			saveState(state);

			try {
				const response = await fetch(
					`https://2xgcgzgjq7.execute-api.us-east-1.amazonaws.com/api/v1/chat`,
					{
						method: "POST",
						headers: {
							"Content-Type": "application/json",
							"x-project-id": `${projectId}`,
							"x-session-id": state.sessionId,
						},
						body: JSON.stringify({
							messages: state.messages,
							website_url: window.location.href,
						}),
					}
				);

				if (!response.ok) {
					const errorText = response.statusText;
					if (response.status === 429) {
						const errorMessage = await response.json();
						if (errorMessage.error.includes("Session Rate limit")) {
							// Handle rate limit - temporarily disable for 2 seconds
							setTimeout(() => {
								input.disabled = false;
								sendButton.disabled = false;
								input.focus();
							}, 2000);
							throw new Error(
								"Please wait a few seconds before sending another message"
							);
						} else {
							// Handle usage limits (both session and project)
							input.disabled = true;
							sendButton.disabled = true;
							if (
								errorMessage.error.includes(
									"Session Usage limit"
								)
							) {
								throw new Error(
									"The quota for messaging has been reached for this session."
								);
							} else {
								throw new Error(
									"The quota for messaging has been reached for this project. Please contact the administrator to request a project quota increase."
								);
							}
						}
					}
					throw new Error(errorText);
				}

				const data = await response.json();
				const { response: botText, sources } = data;

				const botResponse = {
					role: "assistant",
					content: botText,

					timestamp: new Date().toISOString(),
					sources,
				};

				state.messages.push(botResponse);
				saveState(state);

				const botMessageDiv = createMessageElement(
					botResponse,
					state.messages.length - 1
				);
				messagesContainer.appendChild(botMessageDiv);
				messagesContainer.scrollTop = messagesContainer.scrollHeight;
			} catch (error) {
				if (!error.message  || error.message == "Failed to fetch"){
					error.message =
						"I'm having trouble connecting. Please try again later.";
				}

				// Handle error
				const errorResponse = {
					role: "assistant",
					content: error.message,
					timestamp: new Date().toISOString(),
					isError: true,
				};

				state.messages.push(errorResponse);
				saveState(state);

				const errorMessageDiv = createMessageElement(
					errorResponse,
					state.messages.length - 1
				);
				messagesContainer.appendChild(errorMessageDiv);
				messagesContainer.scrollTop = messagesContainer.scrollHeight;
			} finally {
				loadingSpinner.style.display = "none";
				input.disabled = false;
				sendButton.disabled = false;

				// Reset UI state only if not permanently disabled
				//if (!error?.message?.includes("quote for messaging has been reached")) {
				//	loadingSpinner.style.display = 'none';
				//}
			}
		}
	}

	function createMessageElement(msg, idx) {
		const messageDiv = document.createElement("div");

		// Create header with role and timestamp
		const headerDiv = document.createElement("div");
		headerDiv.className = "message-header";

		roleSpan.textContent =
			msg.role.charAt(0).toUpperCase() + msg.role.slice(1);

		const separatorSpan = document.createElement("span");
		separatorSpan.className = "message-header-timestamp";
		separatorSpan.textContent = " • ";

		const timestampSpan = document.createElement("span");
		timestampSpan.className = "message-header-timestamp";
		timestampSpan.textContent = new Date(msg.timestamp).toLocaleTimeString(
			[],
			{
				hour: "numeric",
				minute: "2-digit",
			}
		);

		headerDiv.appendChild(roleSpan);
		headerDiv.appendChild(separatorSpan);
		headerDiv.appendChild(timestampSpan);

		// Create wrapper for message content
		const wrapperDiv = document.createElement("div");
		wrapperDiv.className = "message-wrapper";

		const contentDiv = document.createElement("div");
		contentDiv.className = "message-content thin-scrollbar";

		const textDiv = document.createElement("div");
		textDiv.className = "text-sm markdown-body";
		textDiv.setAttribute("aria-live", "polite"); // or 'assertive' if more urgent

		textDiv.innerHTML = markedFn.parse(msg.content);

		contentDiv.appendChild(textDiv);

		if (msg.role === "assistant" && msg.type !== "fixed") {
			const feedbackButtons = createFeedbackButtons(idx, msg);
			contentDiv.appendChild(feedbackButtons);
		}

		wrapperDiv.appendChild(contentDiv);

		// Add both header and wrapper to main message div
		messageDiv.appendChild(headerDiv);
		messageDiv.appendChild(wrapperDiv);

		return messageDiv;
	}
	function renderSources(sources) {
		return `
			 <div class="sources-container">
            <div class="sources-inner">
				<div class="sources-title">Sources:</div>
				<div class="sources-list">
					${sources
						.map((source) => {
							let displayUrl = source;
							if (displayUrl.endsWith("/")) {
								displayUrl = displayUrl.slice(0, -1);
							}

							if (displayUrl.length > 50) {
								// Check if there's a path component after the last slash
								const lastSlashIndex =
									displayUrl.lastIndexOf("/");
								if (lastSlashIndex !== -1) {
									const lastPathSegment =
										displayUrl.substring(lastSlashIndex);
									displayUrl = lastPathSegment;

									// If still too long, show the last 47 characters
									if (displayUrl.length > 50) {
										displayUrl =
											displayUrl.substring(0, 47) + "...";
									}
								} else {
									// No slash in pathname, show the last 47 characters
									displayUrl =
										"..." +
										displayUrl.substring(
											displayUrl.length - 47
										);
								}
							}

							return `
							<div class="source-item">
								<svg class="source-icon" fill="none" stroke="currentColor" viewBox="0 0 24 24">
									<path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M10 6H6a2 2 0 00-2 2v10a2 2 0 002 2h10a2 2 0 002-2v-4M14 4h6m0 0v6m0-6L10 14" />
								</svg>
								<a href="${source}" target="_blank" rel="noopener noreferrer" class="source-link" title="${source}">
									${displayUrl}
								</a>
							</div>
						`;
						})
						.join("")}
				</div>
			</div>
			</div>
		`;
	}
	function toggleSources(idx) {
		const messageDiv = shadow.querySelectorAll(".message")[idx];
		const existingSourcesContainer =
			messageDiv.querySelector(".sources-container");

			existingSourcesContainer.classList.toggle("visible");

			const msg = state.messages[idx];
			const sourcesContainer = document.createElement("div");
			sourcesContainer.className = "sources-container";
			sourcesContainer.innerHTML = renderSources(msg.sources);
			messageDiv
				.querySelector(".message-content")
				.appendChild(sourcesContainer);

			// Trigger reflow to enable animation
			sourcesContainer.offsetHeight;
			sourcesContainer.classList.add("visible");
		}

		// Trigger active on sources button
		const feedbackButtons = messageDiv.querySelectorAll(".feedback-button");
		feedbackButtons[feedbackButtons.length - 1].classList.toggle(
			"sources-active"
		);

		state.openSourcesIdx = state.openSourcesIdx === idx ? null : idx;
		saveState(state);
	}

			feedbackButtons[0].classList.add("like-active");
		} else if (type === "dislike") {
			feedbackButtons[1].classList.add("dislike-active");
		}
		saveState(state);

		const messagesToFeedback = state.messages.slice(0, idx + 1);
		fetch(
			`https://jly98nq0ri.execute-api.us-east-1.amazonaws.com/feedback/${projectId}`,
			{
				method: "POST",
				headers: {
					"Content-Type": "application/json",
				},
				body: JSON.stringify({
					messages: messagesToFeedback,
					feedback: type,
				}),
			}
		);
	}

				const response = await submitIssue(
					projectId,
					issueText,
					state.messages,
					email
				);

				// Add message with createMessageElement
				const messageDiv = createMessageElement(
					{

				textarea.value = "";

	function debounce(func, delay) {
		let timeout;
		return function (...args) {
			const context = this;
			clearTimeout(timeout);
			timeout = setTimeout(() => func.apply(context, args), delay);
		};
	}

	function pathMatchesPattern(path, pattern) {
		// Convert wildcard pattern to regex
		const regexPattern = pattern
			.replace(/\*/g, ".*") // Replace * with .*
			.replace(/\//g, "\\/"); // Escape forward slashes

		const regex = new RegExp(regexPattern);
		return regex.test(path);
	}

	// Function to check if the chat should be visible based on current URL
	function shouldShowChat() {
		const currentPath = window.location.pathname;

		// If excludePaths is specified and current path matches any pattern, hide chat
		if (excludePathsArray && excludePathsArray.length > 0) {
			for (const pattern of excludePathsArray) {
				if (pathMatchesPattern(currentPath, pattern)) {
					return false;
				}
			}
		}

		// If includePaths is specified, show chat only if current path matches any pattern
		if (includePathsArray && includePathsArray.length > 0) {
			for (const pattern of includePathsArray) {
				if (pathMatchesPattern(currentPath, pattern)) {
					return true;
				}
			}
			// If includePaths is specified but no patterns match, hide chat
			return false;
		}

		// If no paths are specified or no exclude matches found, show chat
		return true;
	}

	let lastPath = "";
	function handleUrlChange() {
		const currentPath = window.location.pathname;

		if (currentPath !== lastPath) {
			lastPath = currentPath;
			const chatContainer = document.getElementById("chat-shadow-root");
			const isVisible = shouldShowChat();

			if (chatContainer) {
				chatContainer.style.display = isVisible ? "block" : "none";
			}
		}
	}

	// Set up URL change detection
	if (window.history && window.history.pushState) {
		window.addEventListener("popstate", handleUrlChange);

		// Intercept pushState and replaceState calls
		const originalPushState = window.history.pushState;
		const originalReplaceState = window.history.replaceState;

		window.history.pushState = function () {
			originalPushState.apply(this, arguments);
			handleUrlChange();
		};

		window.history.replaceState = function () {
			originalReplaceState.apply(this, arguments);
			handleUrlChange();
		};
	}
